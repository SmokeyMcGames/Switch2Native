cmake_minimum_required(VERSION 3.24)
project(Switch2Native VERSION 0.3.9 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(switch2_core src/usb.cpp src/ble.cpp src/protocol.cpp src/motion.cpp src/packed_motion.cpp src/session.cpp src/joy_protocol.cpp src/joy_session.cpp src/joy_mouse.cpp src/gyro_motion.cpp src/windows_pairing.cpp src/controller_settings.cpp src/motion_orientation.cpp)
target_include_directories(switch2_core PUBLIC src)
target_compile_definitions(switch2_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(switch2_core PRIVATE /W4 /permissive- /EHsc)
target_link_libraries(switch2_core PUBLIC setupapi hid winusb windowsapp advapi32)
add_executable(Switch2Native src/main.cpp)
target_link_libraries(Switch2Native PRIVATE switch2_core)
set(WINDOWS_KITS "C:/Program Files (x86)/Windows Kits/10" CACHE PATH "Windows SDK/WDK root")
set(WDK_VERSION "10.0.26100.0" CACHE STRING "Installed WDK version")
set(UMDF_VERSION "2.31" CACHE STRING "UMDF target version")
configure_file("${WINDOWS_KITS}/Include/${WDK_VERSION}/km/hidport.h" "${CMAKE_CURRENT_BINARY_DIR}/wdk/hidport.h" COPYONLY)
add_library(Switch2NativeDriver SHARED driver/driver.cpp)
target_include_directories(Switch2NativeDriver SYSTEM PRIVATE "${WINDOWS_KITS}/Include/wdf/umdf/${UMDF_VERSION}")
target_include_directories(Switch2NativeDriver PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/wdk")
target_compile_definitions(Switch2NativeDriver PRIVATE UMDF_VERSION_MAJOR=2 UMDF_VERSION_MINOR=31 UMDF_USING_NTSTATUS)
target_compile_options(Switch2NativeDriver PRIVATE /W4 /EHsc)
target_link_libraries(Switch2NativeDriver PRIVATE switch2_core "${WINDOWS_KITS}/Lib/wdf/umdf/x64/${UMDF_VERSION}/WdfDriverStubUm.lib" ntdll advapi32)
set_target_properties(Switch2NativeDriver PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/out/driver" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/out/driver")
add_custom_command(TARGET Switch2NativeDriver POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/driver/Switch2Native.inf" "$<TARGET_FILE_DIR:Switch2NativeDriver>/Switch2Native.inf")
set_property(TARGET Switch2NativeDriver APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/driver/Switch2Native.inf")
add_library(Switch2SensorDriver SHARED driver/sensors.cpp)
target_include_directories(Switch2SensorDriver SYSTEM PRIVATE
    "${WINDOWS_KITS}/Include/wdf/umdf/${UMDF_VERSION}"
    "${WINDOWS_KITS}/Include/${WDK_VERSION}/um/sensors/1.1")
target_compile_definitions(Switch2SensorDriver PRIVATE UMDF_VERSION_MAJOR=2 UMDF_VERSION_MINOR=31 UMDF_USING_NTSTATUS)
target_compile_options(Switch2SensorDriver PRIVATE /W4 /permissive- /EHsc)
target_link_libraries(Switch2SensorDriver PRIVATE switch2_core
    "${WINDOWS_KITS}/Lib/wdf/umdf/x64/${UMDF_VERSION}/WdfDriverStubUm.lib"
    "${WINDOWS_KITS}/Lib/${WDK_VERSION}/um/x64/sensors/1.1/sensorscxstub.lib"
    SensorsUtils propsys ole32 ntdll)
set_target_properties(Switch2SensorDriver PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/out/driver")
add_custom_command(TARGET Switch2SensorDriver POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${CMAKE_CURRENT_SOURCE_DIR}/driver/Switch2Sensors.inf" "$<TARGET_FILE_DIR:Switch2SensorDriver>/Switch2Sensors.inf")
set_property(TARGET Switch2SensorDriver APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/driver/Switch2Sensors.inf")
